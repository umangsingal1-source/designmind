-- DesignMind Supabase Schema v2 -- Run this in: supabase.com → your project → SQL Editor → New query → Run -- ============================================================ CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; -- ============================================================ -- 1. PROFILES -- ============================================================ CREATE TABLE public.profiles ( id UUID REFERENCES auth.users(id) ON DELETE CASCADE PRIMARY KEY, phone TEXT UNIQUE, name TEXT, name_id TEXT UNIQUE, company TEXT, position TEXT, company_email TEXT, email TEXT, insta TEXT, location TEXT, website TEXT, birthday TEXT, exp TEXT, avatar_url TEXT, portfolio_token TEXT UNIQUE, status TEXT DEFAULT 'approved', mode TEXT DEFAULT 'full', joined_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users can view own profile" ON public.profiles FOR SELECT USING (auth.uid() = id); CREATE POLICY "Users can update own profile" ON public.profiles FOR UPDATE USING (auth.uid() = id); CREATE POLICY "Users can insert own profile" ON public.profiles FOR INSERT WITH CHECK (auth.uid() = id); -- ============================================================ -- 2. PROJECTS -- ============================================================ CREATE TABLE public.projects ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL, name TEXT NOT NULL, type TEXT, icon TEXT DEFAULT '🏠', label TEXT, color TEXT DEFAULT 'gold', client TEXT, client_phone TEXT, client_email TEXT, loc TEXT, bud TEXT, phase TEXT DEFAULT 'Active', pct INTEGER DEFAULT 0, completed BOOLEAN DEFAULT FALSE, completed_at TIMESTAMPTZ, share_token TEXT UNIQUE, sections JSONB DEFAULT '[]', rooms JSONB DEFAULT '[]', created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.projects ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own projects" ON public.projects FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Public read via share token" ON public.projects FOR SELECT USING (share_token IS NOT NULL); -- ============================================================ -- 3. RENDERS -- ============================================================ CREATE TABLE public.renders ( id TEXT PRIMARY KEY, project_id TEXT REFERENCES public.projects(id) ON DELETE CASCADE, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, section TEXT, note TEXT, status TEXT DEFAULT 'pending', image_url TEXT, image_path TEXT, comments JSONB DEFAULT '[]', tags JSONB DEFAULT '{}', uploaded_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.renders ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own renders" ON public.renders FOR ALL USING (auth.uid() = user_id); CREATE INDEX idx_renders_project ON public.renders(project_id); -- ============================================================ -- 4. BOQ ITEMS -- ============================================================ CREATE TABLE public.boq_items ( id TEXT PRIMARY KEY, project_id TEXT REFERENCES public.projects(id) ON DELETE CASCADE, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, room TEXT, item TEXT, item_desc TEXT, qty NUMERIC, size TEXT, rate NUMERIC DEFAULT 0, total NUMERIC DEFAULT 0, vendor TEXT, status TEXT DEFAULT 'Pending', order_date TEXT, cost NUMERIC DEFAULT 0, notes TEXT, sort_order INTEGER DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.boq_items ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own BOQ" ON public.boq_items FOR ALL USING (auth.uid() = user_id); CREATE INDEX idx_boq_project ON public.boq_items(project_id); -- ============================================================ -- 5. BOQ VENDOR PAYMENTS -- ============================================================ CREATE TABLE public.boq_vendor_payments ( id TEXT PRIMARY KEY, boq_item_id TEXT REFERENCES public.boq_items(id) ON DELETE CASCADE, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, paid_by TEXT, amt NUMERIC, pay_date TEXT, mode TEXT, ref TEXT, note TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.boq_vendor_payments ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own vendor payments" ON public.boq_vendor_payments FOR ALL USING (auth.uid() = user_id); -- ============================================================ -- 6. BOQ CLIENT PAYMENTS -- ============================================================ CREATE TABLE public.boq_client_payments ( id TEXT PRIMARY KEY, project_id TEXT REFERENCES public.projects(id) ON DELETE CASCADE, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, pay_type TEXT, amt NUMERIC, pay_date TEXT, mode TEXT, ref TEXT, notes TEXT, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.boq_client_payments ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own client payments" ON public.boq_client_payments FOR ALL USING (auth.uid() = user_id); -- ============================================================ -- 7. SNAGS -- ============================================================ CREATE TABLE public.snags ( id TEXT PRIMARY KEY, project_id TEXT REFERENCES public.projects(id) ON DELETE CASCADE, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, title TEXT NOT NULL, cat TEXT, priority TEXT DEFAULT 'Medium', snag_desc TEXT, status TEXT DEFAULT 'open', created_by TEXT, resolved_at TIMESTAMPTZ, resolved_by TEXT, comments JSONB DEFAULT '[]', photo_urls JSONB DEFAULT '[]', created_at TIMESTAMPTZ DEFAULT NOW(), updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.snags ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own snags" ON public.snags FOR ALL USING (auth.uid() = user_id); CREATE INDEX idx_snags_project ON public.snags(project_id); -- ============================================================ -- 8. TODOS -- ============================================================ CREATE TABLE public.todos ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, project_id TEXT REFERENCES public.projects(id) ON DELETE SET NULL, todo_text TEXT NOT NULL, done BOOLEAN DEFAULT FALSE, due_date TEXT, priority TEXT DEFAULT 'Normal', created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.todos ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own todos" ON public.todos FOR ALL USING (auth.uid() = user_id); -- ============================================================ -- 9. EXPENSES -- ============================================================ CREATE TABLE public.expenses ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, exp_type TEXT NOT NULL, amount NUMERIC NOT NULL, note TEXT, project_id TEXT REFERENCES public.projects(id) ON DELETE SET NULL, created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.expenses ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own expenses" ON public.expenses FOR ALL USING (auth.uid() = user_id); -- ============================================================ -- 10. PORTFOLIO -- ============================================================ CREATE TABLE public.portfolio_projects ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE, proj_id TEXT, title TEXT, proj_type TEXT, area TEXT, year TEXT, proj_desc TEXT, style TEXT, cover_url TEXT, photo_urls JSONB DEFAULT '[]', created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.portfolio_projects ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users manage own portfolio" ON public.portfolio_projects FOR ALL USING (auth.uid() = user_id); CREATE POLICY "Public can view portfolio" ON public.portfolio_projects FOR SELECT USING (TRUE); -- ============================================================ -- 11. ADMIN CURATED LIBRARY -- ============================================================ CREATE TABLE public.admin_curated ( id SERIAL PRIMARY KEY, pexels_id INTEGER UNIQUE NOT NULL, room TEXT, styles JSONB DEFAULT '[]', mood TEXT, tags JSONB DEFAULT '[]', added_by UUID REFERENCES auth.users(id), added_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.admin_curated ENABLE ROW LEVEL SECURITY; CREATE POLICY "Anyone can read curated" ON public.admin_curated FOR SELECT USING (TRUE); CREATE POLICY "Only admin can write curated" ON public.admin_curated FOR ALL USING ( auth.uid() IN (SELECT id FROM public.profiles WHERE phone = '9836266894') ); -- ============================================================ -- 12. SIGNUP LEADS -- ============================================================ CREATE TABLE public.signup_leads ( id SERIAL PRIMARY KEY, name TEXT, phone TEXT UNIQUE, name_id TEXT, company TEXT, position TEXT, company_email TEXT, status TEXT DEFAULT 'pending', created_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE public.signup_leads ENABLE ROW LEVEL SECURITY; CREATE POLICY "Anyone can submit lead" ON public.signup_leads FOR INSERT WITH CHECK (TRUE); CREATE POLICY "Admin can read leads" ON public.signup_leads FOR SELECT USING ( auth.uid() IN (SELECT id FROM public.profiles WHERE phone = '9836266894') ); -- ============================================================ -- AUTO-UPDATE TRIGGERS -- ============================================================ CREATE OR REPLACE FUNCTION update_updated_at() RETURNS TRIGGER AS $$ BEGIN NEW.updated_at = NOW(); RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER projects_updated_at BEFORE UPDATE ON public.projects FOR EACH ROW EXECUTE FUNCTION update_updated_at(); CREATE TRIGGER renders_updated_at BEFORE UPDATE ON public.renders FOR EACH ROW EXECUTE FUNCTION update_updated_at(); CREATE TRIGGER boq_items_updated_at BEFORE UPDATE ON public.boq_items FOR EACH ROW EXECUTE FUNCTION update_updated_at(); CREATE TRIGGER snags_updated_at BEFORE UPDATE ON public.snags FOR EACH ROW EXECUTE FUNCTION update_updated_at();
